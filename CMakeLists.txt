cmake_minimum_required(VERSION 3.20)
project(sedimentdb VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

option(SEDIMENT_BUILD_TESTS "Build tests" ON)
option(SEDIMENT_BUILD_BENCH "Build db_bench (needs LevelDB for the comparison)" ON)
option(SEDIMENT_WERROR "Warnings as errors" OFF)
set(SEDIMENT_SANITIZE "" CACHE STRING "e.g. address,undefined or thread")

find_package(Threads REQUIRED)
set(WARN -Wall -Wextra -Wpedantic -Wshadow -Wconversion -Wno-sign-conversion)
if(SEDIMENT_WERROR)
  list(APPEND WARN -Werror)
endif()
if(SEDIMENT_SANITIZE)
  add_compile_options(-fsanitize=${SEDIMENT_SANITIZE} -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=${SEDIMENT_SANITIZE})
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
  # Enables the ARMv8 CRC32C instructions used by src/coding.cc.
  add_compile_options(-march=armv8-a+crc)
endif()

add_library(sediment
  src/coding.cc src/env.cc src/log.cc src/memtable.cc src/iterator.cc
  src/table.cc src/version.cc src/write_batch.cc src/db_impl.cc)
target_include_directories(sediment PUBLIC include PRIVATE src)
target_compile_options(sediment PRIVATE ${WARN})
target_link_libraries(sediment PUBLIC Threads::Threads)

add_executable(sediment-crashtest tools/crashtest.cc)
target_link_libraries(sediment-crashtest PRIVATE sediment)
target_include_directories(sediment-crashtest PRIVATE src)
target_compile_options(sediment-crashtest PRIVATE ${WARN})

include(FetchContent)
if(SEDIMENT_BUILD_TESTS)
  enable_testing()
  FetchContent_Declare(googletest
    URL https://github.com/google/googletest/archive/refs/tags/v1.17.0.tar.gz
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
  FetchContent_MakeAvailable(googletest)
  add_executable(sediment_tests tests/format_test.cc tests/table_test.cc tests/db_test.cc tests/model_test.cc)
  target_include_directories(sediment_tests PRIVATE src)
  target_link_libraries(sediment_tests PRIVATE sediment GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(sediment_tests DISCOVERY_TIMEOUT 60)
  add_test(NAME crash_recovery COMMAND sediment-crashtest --rounds 20 --dir ${CMAKE_BINARY_DIR}/crashtest)
endif()

if(SEDIMENT_BUILD_BENCH)
  find_path(LEVELDB_INCLUDE leveldb/db.h HINTS /opt/homebrew/opt/leveldb/include /usr/local/opt/leveldb/include)
  find_library(LEVELDB_LIB leveldb HINTS /opt/homebrew/opt/leveldb/lib /usr/local/opt/leveldb/lib)
  find_package(SQLite3)
  add_executable(db_bench tools/db_bench.cc)
  target_link_libraries(db_bench PRIVATE sediment)
  target_include_directories(db_bench PRIVATE src)
  if(LEVELDB_INCLUDE AND LEVELDB_LIB)
    target_compile_definitions(db_bench PRIVATE HAVE_LEVELDB=1)
    target_include_directories(db_bench PRIVATE ${LEVELDB_INCLUDE})
    target_link_libraries(db_bench PRIVATE ${LEVELDB_LIB})
  endif()
  if(SQLite3_FOUND)
    target_compile_definitions(db_bench PRIVATE HAVE_SQLITE=1)
    target_link_libraries(db_bench PRIVATE SQLite::SQLite3)
  endif()
endif()
